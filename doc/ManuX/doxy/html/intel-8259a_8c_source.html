55/*
56 * Initialisation des PICs.
57 *
58 * intBase stipule le numéro d'interruption correspondant à l'IRQ 0 du maître.
59 * pour l'esclave, c'est en suivant (donc intBase + 8).
60 */
61void i8259aInit(uint8_t intBase)
62{
63 uint8_t m, s;
64 int i;
65
66 // On sauve les masques
67 inb(PIC_MAITRE_PORT_DONNEE, m);
68 inb(PIC_ESCLAVE_PORT_DONNEE, s);
69
70 // On envoie les 4 mots d'initialisation au maître
71
72 // Premier mot ICW1 : initialisation (ICW1_INIT) en 4 mots (ICW1_ICW4)
73 outb(PIC_MAITRE_PORT_COMMANDE, ICW1_INIT | ICW1_ICW4);
74 delaiOutB;
75
76 // Deuxième mot : la base des INT
77 outb(PIC_MAITRE_PORT_DONNEE, intBase);
78 delaiOutB;
79
80 // Troisième mot : le second est en cascade sur IRQ 2
81 outb(PIC_MAITRE_PORT_DONNEE, ICW3_MAITRE2);
82 delaiOutB;
83
84 // Quatrième mot : config de base
85 outb(PIC_MAITRE_PORT_DONNEE, ICW4_8086);
86
87 // Premier mot ICW1 : initialisation (ICW1_INIT) en 4 mots (ICW1_ICW4)
88 outb(PIC_ESCLAVE_PORT_COMMANDE, ICW1_INIT | ICW1_ICW4);
89 delaiOutB;
90
91 // Deuxième mot : la base des INT
92 outb(PIC_ESCLAVE_PORT_DONNEE, intBase + 8);
93 delaiOutB;
94
95 // Troisième mot :il est en cascade du maître
96 outb(PIC_ESCLAVE_PORT_DONNEE, ICW3_ESCLAVE2);
97 delaiOutB;
98
99 // Quatrième mot : config de base
100 outb(PIC_ESCLAVE_PORT_DONNEE, ICW4_8086);
101 delaiOutB;
102
103 // On restaure les masques
104 outb(PIC_MAITRE_PORT_DONNEE, m);//|0xff);
105 outb(PIC_ESCLAVE_PORT_DONNEE, s);//|0xff);
106
107 // On met à NULL les handlers
108 for (i = 0; i < I8259A_NB_IRQ; i++) {
109 nbHandlerClients[i] = 0;
110 // handlerClients[i].handler = (void (*)(void *)) NULL;
111 // handlerClients[i].private = NULL;
112 }
113}

118int i8259aAjouterHandler(int numIRQ, void (*handler)(void *), void * private)
119{
120 // On ne gère que un handler par IRQ
121 if (nbHandlerClients[numIRQ] >= MANUX_NB_HANDLER_PAR_IRQ) {
122 return ENOMEM;
123 }
124
125 handlerClients[numIRQ][nbHandlerClients[numIRQ]].handler = handler;
126 handlerClients[numIRQ][nbHandlerClients[numIRQ]].private = private;
127
128 nbHandlerClients[numIRQ] ++;
129
130 return ESUCCES;
131}

136void i8259aInterdireIRQ(uint8_t numIRQ)
137{
138 char masque;
139 uint16_t port = PIC_MAITRE_PORT_DONNEE;
140
141 // Circuit esclave (0x0f > numIRQ >= 0x08) ?
142 if (numIRQ & 0x08) {
143 port = PIC_ESCLAVE_PORT_DONNEE;
144 numIRQ &= 0x07; // On ne garde que les 3 derniers bits
145 }
146
147 // On lit le masque actuel
148 inb(port, masque);
149
150 // On positionne le masque modifié
151 outb(port, masque & (0x01 << numIRQ));
152}

181void i8259aAckIRQ(uint8_t numIRQ)
182{
183 // Si c'est suite à une interruption de l'esclave il faut *aussi*
184 // prévenir le maître (cascade)
185 if (numIRQ & 0x08) {
186 outb(PIC_ESCLAVE_PORT_COMMANDE, PIC_EOI);
187 }
188 outb(PIC_MAITRE_PORT_COMMANDE, PIC_EOI);
189}
