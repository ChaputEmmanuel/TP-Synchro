277static void consoleTraiterAEC_CSI(Console * cons, int n, int m, char c)
278{
279 switch (c) {
280 case 'F' : // CPL : Cursor beginning of n(=1) lines up
281 cons->colonne = 0;
282 [[fallthrough]];
283 case 'A' : // CUU : Cursor up n (default n=1) cell
284 n = n?n:1;
285 cons->ligne = MAX(cons->ligne - n, 0);
286 break;
287 case 'E' : // CNL : Cursor beginning of n(=1) lines down
288 cons->colonne = 0;
289 [[fallthrough]];
290 case 'B' : // CUD : Cursor down n (default n=1) cell
291 n = n?n:1;
292 cons->ligne = MIN(cons->ligne + n, cons->nbLignes);
293 break;
294 case 'C' : // CUF : Cursor forward n (default n=1) cell
295 n = n?n:1;
296 cons->colonne = MIN(cons->colonne + n, cons->nbColonnes);
297 break;
298 case 'D' : // CUB : Cursor back n (default n=1) cell
299 n = n?n:1;
300 cons->colonne = MAX(cons->colonne - n, 0);
301 break;
302 case 'G' : // CHA: Cursor horizontal absolute n (default n=1)
303 n = n?n:1;
304 cons->colonne = MIN(cons->nbColonnes, n);
305 break;
306 case 'H' : // CUP: Cursor position n, m (default 1)
307 n = n?n:1 - 1; // Ca commence à 1 mais moi à 0
308 m = m?m:1 - 1;
309 cons->ligne = MIN(cons->nbLignes, n);
310 cons->colonne = MIN(cons->nbColonnes, m);
311 break;
312 case 'J' : // ED : Erase in Display
313 if (n == 0) {
314 // On efface tout à partir du curseur
315 } else if (n == 1) {
316 // On efface tout jusqu'au curseur
317 } else if (n == 2) {
318 consoleEffacer(cons);
319 }
320 break;
321 case 'm' : // SGR : Select Graphic Rendition
322 consoleSelectGraphicRendition(cons, n);
323 break;
324 }
325}

338static int consoleGererAEC(Console * cons, char * msg, int nbOctets)
339{
340 int a=0, b=0; //< Il peut y avoir deux coefficients
341 char c; //< Le code lui-même
342 int r = 0; //< Le nombre d'octets consommés
343
344 // On teste et on consomme l'ASCII_ESC
345 if (msg[r] != ASCII_ESC) {
346 goto fin;
347 }
348 r++;
349 if (r >= nbOctets){
350 goto fin;
351 }
352 // Pour le moment on ne gère que les CSI
353 if (msg[r] != '[') {
354 goto fin;
355 }
356 r++;
357
358 // Lecture du premier coefficient
359 while ((r < nbOctets)&&(msg[r] >= ASCII_0)&&(msg[r] <= ASCII_9)) {
360 a = 10*a + msg[r] - ASCII_0;
361 r++;
362 }
363 if (r >= nbOctets){
364 goto fin;
365 }
366 if (msg[r] == ASCII_SEMICOLON) {
367 r++;
368 }
369 if (r >= nbOctets){
370 goto fin;
371 }
372 // Lecture du second coefficient
373 while ((r < nbOctets)&&(msg[r] >= ASCII_0)&&(msg[r] <= ASCII_9)) {
374 b = 10*b + msg[r] - ASCII_0;
375 r++;
376 }
377 if (r >= nbOctets){
378 goto fin;
379 }
380 // On lit enfin le code lui-même
381 c = msg[r];
382
383 // On peut maintenant appliquer le code
384 consoleTraiterAEC_CSI(cons, a, b, c);
385
386 // On termine en donnant le nombre d'octets consommés
387 fin :
388 return r;
389}

391void consoleAfficherN(Console * cons, char * msg, int nbOctets)
392{
393 int r;
394
395 assert(nbOctets > 0);
396
397#ifdef MANUX_CONSOLE_AVEC_MUTEX
398 exclusionMutuelleEntrer(&cons->scAcces);
399#endif
400
401 assert(cons->nbColonnes != 0);
402
403 while (nbOctets) {
404 switch (*msg) {
405 case ASCII_LINE_FEED : // '\n'
406 avancerLigne(cons);
407 [[fallthrough]];
408 case ASCII_CARRIAGE_RETURN : // '\r'
409 cons->colonne = 0;
410 break;
411 case ASCII_ESC :
412 r = consoleGererAEC(cons, msg, nbOctets);
413 msg += r;
414 nbOctets -= r;
415 break;
416 default :
417 consoleAfficherCaractere(cons, *msg);
418 break;
419 }
420 msg++;nbOctets--;
421 }
422#ifdef MANUX_CONSOLE_AVEC_MUTEX
423 exclusionMutuelleSortir(&cons->scAcces);
424#endif
425}

647int consoleLire(Console * cons, void * buffer, int nbOctets)
648{
649 // On ne peut pas en lire plus qu'il y en a !
650 uint16_t nb = min(nbOctets, cons->nbCarAttente);
651 uint16_t lu = 0; // Le cumul des lectures
652 uint16_t aLire; // Combien on en lit à chaque passage
653
654 while (lu < nb) {
655 // On lit sur la "fin" du tableau circulaire
656 aLire = min(nb-lu, 4096 - cons->indiceProchainCar);
657 memcpy(buffer+lu,
658 cons->bufferClavier+cons->indiceProchainCar,
659 aLire);
660 // On décompte cette lecture du buffer
661 cons->indiceProchainCar = (cons->indiceProchainCar + aLire); // WARNING FAUX !
662 cons->nbCarAttente = cons->nbCarAttente - aLire;
663
664 lu = lu + aLire;
665 }
666
667 return lu;
668}

676#define EST_CHIFFRE_DEC(x) ((x>= '0') && (x<= '9'))
677#define EST_CHIFFRE_HEX(x) (((x>= '0') && (x<= '9')) ||((x>= 'a') && (x<= 'f')))
678
679int consoleLireEntier(Console * cons)
680{
681 int result = 0;
682
683 do {
684 // On attend qu'il y ai qqchose
685 while (cons->nbCarAttente == 0)
686 ordonnanceur();
687
688 // On lit tant qu'il y a des chiffres et on arrête quand il y a
689 // autre chose
690 while ((cons->nbCarAttente)
691 && (EST_CHIFFRE_HEX(cons->bufferClavier[cons->indiceProchainCar]))
692 ) {
693 if (EST_CHIFFRE_DEC(cons->bufferClavier[cons->indiceProchainCar])) {
694 result = 16 * result + cons->bufferClavier[cons->indiceProchainCar] - '0';
695 } else {
696 result = 16 * result + cons->bufferClavier[cons->indiceProchainCar] - 'a' + 10;
697 }
698 cons->nbCarAttente--;
699 cons->indiceProchainCar++; // WARNING modulo taille !!!
700 }
701 } while ((cons->nbCarAttente == 0)
702 ||(EST_CHIFFRE_HEX(cons->bufferClavier[cons->indiceProchainCar])));
703 // On supprime le caractère qui nous a fait finir (WARNING !!)
704 if (cons->nbCarAttente) {
705 cons->nbCarAttente--;
706 cons->indiceProchainCar++; // WARNING modulo taille !!!
707 }
708 return result;
709}
