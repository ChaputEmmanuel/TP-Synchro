100static
101void debugMasqueAfficher(void)
102{
103 int n;
104 char c;
105 int i = 0;
106
107 masqueDebugageConsoleSave = masqueDebugageConsole;
108 masqueDebugageConsole = 0x00;
109
110 printkc("%c[2J", ASCII_ESC);
111
112 for (n = 0 ; n < 15 ; n++) {
113 debugMasqueAfficherFlag(n);
114 }
115 printkc("%c[20;0H");
116 printkc("masqueDebugageConsole = 0x%x\n", masqueDebugageConsoleSave);
117 printkc("masqueDebugageFichier = 0x%x\n", masqueDebugageFichier);
118
119 printkc("%c[30m", ASCII_ESC);
120 printkc("%c[100m", ASCII_ESC);
121
122 printkc("%c[24;0Hi = Haut / k = bas / j = Console / l = fichier / espace = fin ", ASCII_ESC);
123 printkc("%c[0m", ASCII_ESC);
124
125 do {
126 while (consoleLire(tacheEnCours->console, &c, 1) == 0){}; // WARNING
127 printkc("%c[%d;%dH ", ASCII_ESC, i+2, 4);
128 switch (c) {
129 case 'i' :
130 i--;
131 i = MAX(i, 0);
132 break;
133 case 'k' :
134 i++;
135 i = MIN(i, 14);
136 break;
137 case 'j' :
138 masqueDebugageConsoleSave = masqueDebugageConsoleSave ^(1 << i);
139 debugMasqueAfficherFlag(i);
140 printkc("%c[20;0H");
141 printkc("masqueDebugageConsole = 0x%x\n", masqueDebugageConsoleSave);
142 printkc("masqueDebugageFichier = 0x%x\n", masqueDebugageFichier);
143 break;
144 case 'l' :
145 masqueDebugageFichier = masqueDebugageFichier ^(1 << i);
146 debugMasqueAfficherFlag(i);
147 printkc("%c[20;0H");
148 printkc("masqueDebugageConsole = 0x%x\n", masqueDebugageConsoleSave);
149 printkc("masqueDebugageFichier = 0x%x\n", masqueDebugageFichier);
150 break;
151 }
152 printkc("%c[%d;%dH>", ASCII_ESC, i+2, 4);
153 } while (c != ' ');
154 masqueDebugageConsole = masqueDebugageConsoleSave;
155}

335 buffer[d--] = '\n';

337 buffer[d--] = ecran[2*s];
338 s--;
339 }
340 printk(buffer+d+1);
341
342#ifdef MANUX_KMALLOC
