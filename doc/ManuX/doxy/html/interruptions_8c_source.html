384void gestionExceptionPanique(TousRegistres registres,
385 uint32_t numIt,uint32_t errCode,
386 uint32_t eip, uint32_t cs, uint32_t eFlags)
387{
388 char * ecran = MANUX_CON_SCREEN;
389 int d = 1;
390 unsigned char c;
391
392 ecranDeLaMort(errCode, numIt, registres, eip, cs, eFlags);
393
394 while (1) {
395 inb(0x60, c);
396 switch (c) {
397 case 0x01: // ESC
398#ifdef MANUX_CONSOLES_VIRTUELLES
399 basculerVersConsoleSuivante();
400#endif
401 while (c == 0x81){
402 inb(0x60, c);
403 };
404 break;
405 case 0x39: // SPACE On alterne entre l'écran bleu et l'écran au moment du drame
406 d = 1-d;
407 if (d) {
408 ecranDeLaMort(errCode, numIt, registres, eip, cs, eFlags);
409 }else{
410 // On restaure l'écran
411 memcpy(ecran, bufferEcran, 4000);
412 };
413 while (c == 0x39){
414 inb(0x60, c);
415 };
416 break;
417 }
418 }
419 halt();
420}
