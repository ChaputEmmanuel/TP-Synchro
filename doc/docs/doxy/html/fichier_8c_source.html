189Fichier * fichierDupliquer(Fichier * f)

191 int res;
192 Fichier * result;
193
194 result = kmalloc(sizeof(Fichier));
195
196 if (result == NULL) {
197 paniqueNoyau("Plus de memoire !\n");
198 } else {
199 // On fait une coie simple, ...
200 bcopy(f, result, sizeof(Fichier));
201
202 // ... et on invoque la méthode d'ouverture, si elle a des
203 // décomptes à faire, par exemple
204 if (f->iNoeud->methodesFichier->ouvrir != NULL) {
205 res = f->iNoeud->methodesFichier->ouvrir(f->iNoeud, result, f->fanions, f->mode);
206 if (res != ESUCCES) {
207 kfree(result);
208 result = NULL;
209 }
210 }
211 }
212
213 return result;
