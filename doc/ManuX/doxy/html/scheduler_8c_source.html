95void ordonnanceur(void)
96{
97 Tache * tachePrecedente = tacheEnCours;
98
99 assert(tacheEnCours != NULL);
100
101 printk_debug(DBG_KERNEL_ORDON, "in (de tache %d)\n", tachePrecedente->numero);
102
103#ifdef MANUX_CONSOLES_VIRTUELLES
104 // Basculement entre les consoles virtuelles WARNING pourquoi ici
105 // !? A faire faire par dummyTask
106 if (basculeConsoleDemandee) {
107 basculeConsoleDemandee = FALSE;
108 basculerVersConsoleSuivante();
109 }
110#endif
111
112 // Attention, si la tâche en cours n'est pas préemptible
113 if (tacheEnCours->nonPreemptible) {
114 printk_debug(DBG_KERNEL_ORDON, "abandon (tache %d non preemptible)\n", tachePrecedente->numero);
115 return;
116 }
117
118 // (1) On s'occupe de la tâche en cours
119
120 // On cumule le temps d'écution dont elle vient de profiter
121 tacheEnCours->tempsExecution += (nbTopHorloge - dateDernierOrdonnancement);
122
123 // Si on n'est pas là spontanément, on se considère en cours, mais
124 // c'est fini pour le moment !
125 if (tacheEnCours->etat == Tache_En_Cours) {
126 tacheEnCours->etat = Tache_Prete;
127
128 insererCelluleTache(&listeTachesPretes,
129 tacheEnCours,
130 (CelluleTache*)tacheEnCours+sizeof(Tache));
131 }
132
133 // Dans les autres cas, la tâche est sensée être dans une file
134 // correspondant à son état
135
136 // (2) on cherche la tâche suivante
137 // On prend la première tâche prête, il y en a au moins une : la dummy
138 printk_debug(DBG_KERNEL_ORDON, "on cherche la suivante ...\n");
139 do {
140 tacheEnCours = extraireTache(&listeTachesPretes);
141 } while (tacheEnCours->etat != Tache_Prete);
142 printk_debug(DBG_KERNEL_ORDON, "... ce sera la %d\n", tacheEnCours->numero);
143
144 tacheEnCours->etat = Tache_En_Cours;
145
146 /* On note la date pour pouvoir mesurer le temps dont elle va profiter */
147 dateDernierOrdonnancement = nbTopHorloge;
148
149 if (tacheEnCours != tachePrecedente){
150 printk_debug(DBG_KERNEL_ORDON, "On passe a la tache %d de TSS 0x%x \n",
151 tacheEnCours->numero,
152 tacheEnCours->indiceTSSDescriptor);
153
154 /* Une activation de plus pour elle */
155 tacheEnCours->nbActivations++;
156
157 printk_debug(DBG_KERNEL_ORDON, "out (vers tache %d)\n", tacheEnCours->numero);
158 basculerVersTache(tacheEnCours);
159 }
160 printk_debug(DBG_KERNEL_ORDON, "back (vers tache %d)\n", tacheEnCours->numero);
161}

178void initialiserScheduler(void)
179{
180 Tache * t1; // Le fil d'exécution en cours, qui deviendra init
181#ifdef MANUX_DUMMY_TASK
182 Tache * t2; // Ce sera la "dummy" task
183#endif // MANUX_DUMMY_TASK
184
185 dateDernierOrdonnancement = nbTopHorloge;
186
187 printk_debug(DBG_KERNEL_ORDON, "initialisation du scheduler ...\n");
188
189 // Initialisation de la liste (vide) des tâches en cours
190 initialiserListeTache(&listeTachesPretes);
191
192 // Initialisation de la liste (vide) de toutes les tâches
193 initialiserListeTache(&listeToutesLesTaches);
194
195 // Initialisation de la liste (vide) des tâches terminées
196 initialiserListeTache(&listeTachesTerminees);
197
198 // Création d'une tâche pour le fil actuel (premier numéro)
199 t1 = tacheCreer(NULL);
200 if (t1 == NULL) {
201 paniqueNoyau("impossible de creer la premiere tache !\n");
202 }
203#ifdef MANUX_TACHE_CONSOLE
204 tacheSetConsole(t1, consoleNoyau());
205#endif
206 /* Cas particulier de la première tâche : */
207 /* . on la déclare comme en cours. */
208 tacheEnCours = t1;
209 t1->etat = Tache_En_Cours;
210
211 /* . on charge son task register ; */
212 ltr(t1->indiceTSSDescriptor);
213
214 // Avant de permettre à une deuxième tâche d'entrer en concurrence,
215 // il faut s'assurer qu'on a la main sur le noyau.
216#if !defined(MANUX_REENTRANT) && defined(MANUX_EXCLUSION_MUTUELLE)
217 printk_debug(DBG_KERNEL_ORDON, "on verouille le verrou\n");
218
219 exclusionMutuelleEntrer(&verrouGeneralDuNoyau);
220 assert(tacheDansLeNoyau == 0);
221 tacheDansLeNoyau = tacheEnCours->numero;
222#endif
223
224#ifdef MANUX_DUMMY_TASK
225 printk_debug(DBG_KERNEL_ORDON, "creons la dummy\n");
226
227 // Initialisation de la tache "aDummyKernelTask" (numéro 2)
228 t2 = tacheCreer(aDummyKernelTask);
229
230 if (t2 == NULL) {
231 paniqueNoyau("impossible de creer la seconde tache !\n");
232 }
233
234 printk_debug(DBG_KERNEL_ORDON, "la dummy est la ...\n");
235
236#define MANUX_DUMMY_TASK_CONSOLE
237#ifdef MANUX_TACHE_CONSOLE
238#ifdef MANUX_DUMMY_TASK_CONSOLE
239 tacheSetConsole(t2, creerConsoleVirtuelle());
240#else
241 tacheSetConsole(t2, consoleNoyau());
242#endif
243#endif
244
245 printk_debug(DBG_KERNEL_ORDON, "on ajoute la t2 dans l'ordo\n");
246
247 // A partir de maintenant, nous ne sommes plus seuls
248 ordonnanceurAddTache(t2);
249
250#endif // MANUX_DUMMY_TASK
251
252 printk_debug(DBG_KERNEL_ORDON, "scheduler is done\n");
253}
