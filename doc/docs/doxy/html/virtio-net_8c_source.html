251int virtioNetInitPeripherique(int PCINumeroPeripherique)
252{
253 PCIEquipement * pciEquip = PCIEquipementNumero(PCINumeroPeripherique);
254 // WARNING, va le chercher dans le virtio
255
256 VirtioFileVirtuelle * fr;
257 uint8_t addr;
258 void * pointeur; // Pour allouer de la mémoire en
259 // réception
260
261 // On va construire un tableau de buffers (et un de leurs
262 // longueurs) pour passer à virtioFournirBuffers
263 void * buffers[2];
264 int longueurs[2];
265
266 // On renseigne la structure
267 virtioInitPeripheriquePCI(&(virtioReseau.virtioPeripherique),
268 PCINumeroPeripherique,
269 VIRTIO_NET_F_GUEST_CSUM // C'est pas moi qui check
270 | VIRTIO_NET_F_GUEST_TSO4 // Je ne fais pas de TSO
271 | VIRTIO_NET_F_GUEST_TSO6
272 | VIRTIO_NET_F_GUEST_UFO // Pas d'UFO svp
273 | VIRTIO_NET_F_MRG_RXBUF // Je ne merge pas les buffers
274 | VIRTIO_NET_F_CTRL_VQ); // Pas de canal de contrôle
275
276 // On va tout de suite remplir la file de réception de
277 // buffers ([3] page 26 point 5)
278 fr = &(virtioReseau.virtioPeripherique.filesVirtuelles[0]);
279
280 // ATTENTION ici je tente de passer des trames de taille max uniquement
281 // d'où VIRTIO_TAILLE_TRAME_MAX qui remplace sizeof ...
282
283 // Pas de malloc, ...
284 pointeur = allouerPages(NB_PAGES(fr->taille*(sizeof(VirtioReseauEntete)
285 + VIRTIO_TAILLE_TRAME_MAX)/2));
286
287 // Concrètement, puisque le réseau préfixe systématiquement par un
288 // entête à lui, on va lui fourni systématiquement une chaîne avec
289 // un premier buffer pour l'entête, comme ça la trame sera toute
290 // seule dans son buffer, donc directement manipulable ! Malin JP, ...
291 for (int i = 0 ; i < fr->taille/2; i++) {
292 buffers[0] = pointeur;
293 longueurs[0] = sizeof(VirtioReseauEntete);
294 pointeur += sizeof(VirtioReseauEntete);
295 buffers[1] = pointeur;
296 longueurs[1] = VIRTIO_TAILLE_TRAME_MAX;
297 pointeur += VIRTIO_TAILLE_TRAME_MAX;
298 virtioFournirBuffers(&(virtioReseau.virtioPeripherique),
299 VR_FILE_RECEPTION, // File réception
300 buffers, longueurs,
301 2,
302 VRING_DESC_F_WRITE); // pour recevoir
303 }
304
305 // Lecture de l'adresse IEEE
306 for (int i = 0; i < 6; i++){
307 inb((uint16_t)(pciEquip->adresseES + 0x14 + i), addr);
308 virtioReseau.adresseMAC[i] = addr;
309 }
310 /*
311 printk("MAC@ = ");
312 for (int i = 0 ; i < 5 ; i++) {
313 printk("%x:", virtioReseau.adresseMAC[i]);
314 }
315 printk("%x\n", virtioReseau.adresseMAC[5]);
316 */
317 // On définit notre fonction de gestion des interuptions
318 if (i8259aAjouterHandler(pciEquip->interruption,
319 virtioNetGestionInt,
320 &virtioReseau)){
321 printk(PRINTK_CRITIQUE "Trop de handler sur l'interruption %d\n", pciEquip->interruption);
322 paniqueNoyau("Pas possible");
323 return ENOENT;
324 }
325 i8259aAutoriserIRQ(pciEquip->interruption);
326 virtioReseau.nbItRecues = 0;
327
328 // On l'enregistre auprès du système réseau
329 reseauAjouterInterface(&virtioPiloteReseau, &virtioReseau);
330
331 return ESUCCES;
332}

342 // On va construire un tableau de buffers (et un de leurs

373int virtioNetInit(void)
374{
375 int PCINumeroPeripherique;
376
377 printk_debug(DBG_KERNEL_NET, "IN\n");
378
379 // On s'identifie comme pilote réseau
380 reseauEnregistrerPilote(&virtioPiloteReseau);
381
382 // On va chercher un peripherique virtio net
383 printk_debug(DBG_KERNEL_NET, "Je cherche un PCI vendeur/Id %d/%d\n",
384 PCI_VENDEUR_VIRTIO,PCI_PERIPHERIQUE_VIRTIO_NET);
385 PCINumeroPeripherique = PCIObtenirProchainEquipement(PCI_VENDEUR_VIRTIO,
386 PCI_PERIPHERIQUE_VIRTIO_NET, -1);
387 printk_debug(DBG_KERNEL_NET, "Peripherique PCI %d\n", PCINumeroPeripherique);
388
389 // Initialisation de l'unique périphérique pour le moment
390 if (virtioNetInitPeripherique(PCINumeroPeripherique) == 0) {
391 printk_debug(DBG_KERNEL_NET, "Peripherique initialise !\n");
392 }
393
394 // Test d'émission
395 virtioNetEmettre(&virtioReseau, requeteARP);
396
397 printk_debug(DBG_KERNEL_NET, "out\n");
398 return ESUCCES;
399}
