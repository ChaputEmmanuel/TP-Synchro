46void bootloaderInitialiser(void)
47{
48 // On copie les infos dans la variable infoSysteme, qui est dans
49 // une zone que la gestion mémoire va considérée comme
50 // réservée. Ainsi le contenu est préservé.
51 memcpy(&infoSysteme, _infoSysteme, sizeof(InfoSysteme));
52
53 // Un petit message
54 printk_debug(DBG_KERNEL_BOOTLOADER, "SignatureBootloader = 0x%x, is=0x%x (flags = 0x%x)\n",
55 signatureBootloader,
56 infoSysteme,
57 infoSysteme.flags);
58
59 // Quel est le bootloader ?
60 switch (signatureBootloader) {
61 case MULTIBOOT_SIGNATURE :
62 printk_debug(DBG_KERNEL_BOOTLOADER, "C'est multiboot qui nous a lance.\n");
63 break;
64 case MANUX_INIT_SIGNATURE :
65 printk_debug(DBG_KERNEL_BOOTLOADER, "C'est init-manux qui nous a lance.\n");
66 break;
67 default :
68 paniqueNoyau("Bootloader inconnue : 0x%x\n", signatureBootloader);
69 }
70
71 // Alignement des modules à la page ? (Pas important pour nous)
72 if ((infoSysteme.flags & MULTIBOOT_PAGE_ALIGN) == MULTIBOOT_PAGE_ALIGN) {
73 printk_debug(DBG_KERNEL_BOOTLOADER, "Alignement a la page\n");
74 }
75
76 // A-t-on des infos sur la mémoire ?
77 if ((infoSysteme.flags & MULTIBOOT_MEMORY_INFO) == MULTIBOOT_MEMORY_INFO) {
78 printk_debug(DBG_KERNEL_BOOTLOADER, "Base = %d, etendue = %d\n",
79 infoSysteme.memoireDeBase,
80 infoSysteme.memoireEtendue);
81 } else {
82 paniqueNoyau("Aucune information sur la memoire !\n");
83 }
84
85 // Informations de la ligne de commande (eg GRUB)
86 if ((infoSysteme.flags & MULTIBOOT_INFO_CMDLINE) == MULTIBOOT_INFO_CMDLINE) {
87 printk_debug(DBG_KERNEL_BOOTLOADER, "Command line : \"%s\"\n", infoSysteme.ligneCommande);
88
89 // On sauvegarde ça
90 int l = strlen(infoSysteme.ligneCommande);
91 memcpy(cmdLine, infoSysteme.ligneCommande, ((l > CMDLINE_MAX_LENGTH)?CMDLINE_MAX_LENGTH:l));
92 cmdLine[(l > CMDLINE_MAX_LENGTH)?CMDLINE_MAX_LENGTH:l] = 0;
93 }
94}

104void bootloaderLireLigneCmd(void)
105{
106#ifdef MANUX_REGISTRE
107 char param[512] ; // Une chaîne de caractères qui contient le
108 // paramètre en cours d'analyse
109 int lgParam;

111 char * debut, * fin ; // Pointeurs

113 // La ligne de commande est une liste d'éléments séparés par des
114 // BOOTLOADER_CMD_SEPARATEUR
115
116 debut = cmdLine;
117
118 while (strlen(debut)) {
119 fin = prochainDelimiteur(debut, BOOTLOADER_CMD_SEPARATEUR);
120 assert(fin > debut); // Puisque strlen(debut) != 0
121
122#ifdef MANUX_REGISTRE // A priori on place dans le registre
123 lgParam = fin - debut;
124 memcpy(param, debut, lgParam);
125 param[lgParam] = 0;
126 registreSystemeAjouterC(param);
127#else // Si pas de registre, on gère ce qu'on sait faire
128# ifdef MANUX_DEBUGMASK_VAR
129 if (strncmp(debut, NOM_MASQUE_CONSOLE, strlen(NOM_MASQUE_CONSOLE)) == 0) {
130 masqueDebugageConsole = atoihex(debut + strlen(NOM_MASQUE_CONSOLE)+1);
131 } else if (strncmp(debut, NOM_MASQUE_FICHIER, strlen(NOM_MASQUE_FICHIER)) == 0) {
132 masqueDebugageFichier = atoihex(debut + strlen(NOM_MASQUE_FICHIER)+1);
133 }
134# endif // MANUX_DEBUGMASK_VAR
135#endif // MANUX_REGISTRE
136
137 // Si on a un séparateur, on avance, sinon, on est sur le 0 final
138 if (*fin == BOOTLOADER_CMD_SEPARATEUR) {
139 debut = fin + 1;
140 } else {
141 debut = fin;
142 }
143 }
