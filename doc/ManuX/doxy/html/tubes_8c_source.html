108static
109size_t tubeEcrire(Fichier * f, void * buffer, size_t nbOctets)
110{
111 Tube * tube;
112 uint32_t n = 0;
113 uint32_t nbOctetsEcrits = 0; // Le nombre d'octets déja écrits
114
115 printk_debug(DBG_KERNEL_TUBE, "in\n");
116
117 // Peut-on décemment écrire dans le tube ?
118 if ((f == NULL) || (f->iNoeud == NULL) || (f->iNoeud->prive == NULL)) {
119 return -EINVAL;
120 }
121 tube = f->iNoeud->prive;
122
123 exclusionMutuelleEntrer(&(tube->exclusionMutuelle));
124
125 // On fait une boucle, car il est possible que l'on doive écrire en
126 // deux fois si on est proche de la fin du tableau qui contient les
127 // données.
128 do {
129 // On n'écrit ni plus que ce qui est demandé, ni plus que ce
130 // qu'on peut
131 n = MIN(nbOctets - nbOctetsEcrits, MANUX_TUBE_CAPACITE - tube->taille);
132
133 // On ne doit pas aller écrire au delà du buffer
134 n = MIN(n, (MANUX_TUBE_CAPACITE - tube->indiceProchain));
135
136 // On peut donc copier n octets dans le buffer à partir de la
137 // position courante, sans risque de déborder
138 memcpy(tube->donnees + tube->indiceProchain, buffer, n);
139
140 ordonnanceur(); // Pour forcer
141
142 tube->indiceProchain = (tube->indiceProchain + n) % MANUX_TUBE_CAPACITE;
143 tube->taille += n;
144
145 buffer += n;
146
147 nbOctetsEcrits += n;
148 } while (n > 0);
149
150 if (nbOctetsEcrits > 0) {
151 conditionSignaler(&(tube->nouvellesDonnees));
152 }
153
154 exclusionMutuelleSortir(&(tube->exclusionMutuelle));
155
156 printk_debug(DBG_KERNEL_TUBE, "out\n");
157
158 return nbOctetsEcrits;
159}

161static
162size_t tubeLire(Fichier * f, void * buffer, size_t nbOctets)
163{
164 Tube * tube;
165 int n = 0;
166 int nbOctetsLus = 0;
167 int indicePremier;
168
169 printk_debug(DBG_KERNEL_TUBE, "in\n");
170
171 // Peut-on décemment lire dans le tube ? (note : les deux premières
172 // conditions sont assurées par l'appelant (fichierLire) a priori
173 if ((f == NULL) || (f->iNoeud == NULL) || (f->iNoeud->prive == NULL)) {
174 return -EINVAL;
175 }
176 tube = f->iNoeud->prive;
177
178 exclusionMutuelleEntrer(&(tube->exclusionMutuelle));
179 printk_debug(DBG_KERNEL_TUBE, "mutex locked\n");
180
181 // En cas de lecture bloquante, j'attends la dispo des données ou
182 // la disparition du dernier écrivain
183 if ((f->fanions & O_NONBLOCK) == 0) {
184 while ((tube->taille == 0) && (tube->nbEcrivains > 0)) {
185 printk_debug(DBG_KERNEL_TUBE, "attente condition\n");
186 conditionAttendre(&(tube->nouvellesDonnees), &(tube->exclusionMutuelle));
187 }
188 }
189 // Je sais maintenant qu'il y a des données, ou alors qu'il n'y en
190 // aura plus jamais, ou alors je suis en non bloquant, bref, je
191 // peux y aller et faire comme si, ...
192
193 do {
194 // A partir de quel octet peut-on lire ?
195 indicePremier = (tube->indiceProchain + MANUX_TUBE_CAPACITE - tube->taille)
196 %MANUX_TUBE_CAPACITE;
197
198 // On ne lit ni plus que ce qui est demandé, ni plus que ce
199 // qu'on a
200 n = MIN(nbOctets - nbOctetsLus, tube->taille);
201
202 // On ne doit pas aller lire au delà du buffer
203 n = MIN(n, (MANUX_TUBE_CAPACITE - indicePremier));
204
205 printk_debug(DBG_KERNEL_TUBE,"Je vais lire %d\n", n);
206
207 // On peut donc copier n octets dans le buffer à partir de la
208 // position courante, sans risque de déborder
209 memcpy(buffer, tube->donnees + indicePremier, n);
210
211 ordonnanceur(); // Pour forcer
212
213 indicePremier = (indicePremier + n) % MANUX_TUBE_CAPACITE;
214 tube->taille -= n;
215
216 buffer += n;
217
218 nbOctetsLus += n;
219 } while (n > 0);
220
221 exclusionMutuelleSortir(&(tube->exclusionMutuelle));
222
223 printk_debug(DBG_KERNEL_TUBE, "out\n");
224
225 return nbOctetsLus;
226}
