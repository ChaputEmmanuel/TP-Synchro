258registre * registreAjouterSousRegistre(registre * reg, char * nomSr)
259{
260 registre * result;
261 parametre * sRParam;
262
263 // Existe-t-il déjà ?
264 result = registreChercherSousRegistre(reg, nomSr);
265
266 // Si non, on le crée
267 if (result == NULL) {
268
269 // Création du (sous) reg
270 result = registreCreer(nomSr);
271
272 // On en fait un paramètre
273 sRParam = parametreCreer(nomSr, NULL, (void *) result, NULL);
274 sRParam->estUnRegistre = 1;
275
276 // On l'insère dans le registre père
277 registreAjouterParametre(reg, sRParam);
278 }
279
280 return result;
281}

295static
296void registreAjouterC(registre * reg, char * chaine)
297{
298 char * debut; // Pointeur sur le début du nom
299 char * fin; // Pointeur sur la fin du nom courant
300 char nom[512]; // Une chaine pour héberger temporairement le nom courant
301 int lgNom; // Longueur du nom courant
302 char * valeur;
303 registre * regSS = reg; // La base du sous-système dans lequel on
304 // va introduire
305 parametre * param; // Le parametre qu'on va créer/modifier
306 char * aff = NULL; // Pour restauter le signe d'affectation
307
308 // On fait pointer valeur sur la représentation ou une chaîne vide
309 valeur = prochainDelimiteur(chaine, REGISTRE_AFFECTATION);
310
311 // On s'assure que le nom (qui précède l'affectation) est terminé
312 // par un 0
313 if (*valeur == REGISTRE_AFFECTATION) {
314 *valeur = 0;
315 aff = valeur; // On note où il était
316 valeur ++;
317 }
318
319 // Nous avons donc maintenant deux chaines de caractères (collées
320 // l'une derrière l'autre) l'une contient un nom, l'autre une
321 // valeur (cette dernière peut être une chaîne vide)
322 debut = chaine;
323
324 // Il faut maintenant construire et/ou renseigner la structure sur
325 // la base du nom
326 while (strlen(debut)) {
327 fin = prochainDelimiteur(debut, REGISTRE_NOM_SEPARATEUR);
328 assert(fin > debut); // Puisque strlen(debut) != 0
329 lgNom = fin - debut;
330 memcpy(nom, debut, lgNom);
331 nom[lgNom] = 0;
332 // Si on a un séparateur, on descend dans l'arborescence, sinon,
333 // on est sur le 0 final
334 if (*fin == REGISTRE_NOM_SEPARATEUR) {
335
336 // On descend dans l'arborescence
337 regSS = registreChercherSousRegistre(reg, nom);
338 if (regSS == NULL) {
339 regSS = registreAjouterSousRegistre(reg, nom);
340 }
341 // Eventuellement en la créant
342 if (regSS != NULL) {
343 reg = regSS; // On travaille maintenant dans le sous reg
344 }
345
346 // On passe au suivant
347 debut = fin + 1;
348 // Si on est à la dernière, on peut placer la donnée
349 } else {
350 param = listeParametresChercher(regSS->parametres, nom);
351 if (param == NULL) {
352 param = parametreCreer(nom, valeur, NULL, NULL);
353 listeParametresInserer(regSS->parametres, param);
354 } else {
355 parametreAffecterValeur(param, valeur);
356 }
357
358 // On en arrive à la fin
359 debut = fin;
360 }
361 }
362 // On restaure le symbole d'affectation
363 if (aff) {
364 *aff = REGISTRE_AFFECTATION;
365 }
366}
