19int setDescripteurSegment(DescriptorTable * dt,
20 uint32_t adresse, uint32_t limite,
21 uint8_t type,
22 uint8_t gd0a)
23/*
24 * gd0a doit contenir dans son quadret de poids fort les bits G, D/B, 0
25 * et AVL ([1] p 3-11)
26 */
27{
28 if (dt->taille >= dt->capacite) {
29 return -ENOMEM;
30 }
31 dt->descripteur[dt->taille].ds.limiteFaible = (uint16_t)((limite)&0xFFFF);
32 dt->descripteur[dt->taille].ds.baseFaible = (uint16_t)((adresse)&0xFFFF);
33 dt->descripteur[dt->taille].ds.baseInter = (uint8_t)((adresse>>16)&0xFF);
34 dt->descripteur[dt->taille].ds.type = type;
35 dt->descripteur[dt->taille].ds.limiteFort = (uint8_t)((limite>>16)&0x0F)|gd0a;
36 dt->descripteur[dt->taille].ds.baseFort = (uint8_t)((adresse>>24)&0xFF);
37
38 return (dt->taille++)<<3;
39}

156void initialiserGDT(void)
157{
158 int resultat;
159#define MANUX_GDT_CS 0x08
160#define MANUX_GDT_DS 0x10
161#define MANUX_GDT_SS 0x18
162
163 gdtSysteme = (DescriptorTable *) MANUX_ADRESSE_GDT;
164
165 gdtSysteme->capacite = MANUX_GDT_NB_PAGES * MANUX_TAILLE_PAGE / sizeof(Descripteur);
166 gdtSysteme->taille = 0;
167
168 /* Le segment nul (obligatoire) */
169 resultat = setDescripteurSegment(gdtSysteme, 0, 0, 0, 0);
170 if (resultat != 0x00) {
171 paniqueNoyau("Probleme sur le segment nul en %d\n", resultat);
172 }
173
174 // WARNING pour quoi si je change l'ordre de création (et donc bien
175 // sur le numéro de façon cohérente) cela ne fonctionne pas !?
176 //
177 // Le segment de code qui commmence à l'adresse 0, avec une limite
178 // à 1 méga(pages puisque Gr=1), donc 4G de mémoire.
179 // g00a = 1100 (Gr=1 : blocs de 4K, Sz=1 32 bits protected)
180 // type = 10011010
181 // present, priv=0, code/data, exec, dir up, read, access = 0
182 resultat = setDescripteurSegment(gdtSysteme, 0, 0xFFFFFFFF, 0x9A, 0xC0);
183 if (resultat != MANUX_GDT_CS) {
184 paniqueNoyau("Probleme sur le segment de code en %d\n", resultat);
185 }
186
187 // Le segment de data qui commmence à l'adresse 0, avec une limite
188 // à 1 méga(pages puisque Gr=1), donc 4G de mémoire.
189 // g00a = 1100 (Gr=1 : blocs de 4K, Sz=1 32 bits protected)
190 // type = 10010010
191 // present, priv=0, code/data, noexec, dir up, rw, access = 0
192 resultat = setDescripteurSegment(gdtSysteme, 0, 0xFFFFFFFF, 0x92, 0xC0);
193 if (resultat != MANUX_GDT_DS) {
194 paniqueNoyau("Probleme sur le segment de donnees en %d\n", resultat);
195 }
196
197 // Le segment de pile qui commmence à l'adresse 0, avec une limite
198 // à 1 méga(pages puisque Gr=1), donc 4G de mémoire.
199 // g00a = 1100 (Gr=1 : blocs de 4K, Sz=1 32 bits protected)
200 // type = 10010010
201 // present, priv=0, code/data, noexec, dir up, rw, access = 0
202 resultat = setDescripteurSegment(gdtSysteme, 0, 0xFFFFFFFF, 0x92, 0xC0);
203 if (resultat != MANUX_GDT_SS) {
204 paniqueNoyau("Probleme sur le segment de pile en %d\n", resultat);
205 }
206
207 /* On charge la GDT */
208 chargerGDT(gdtSysteme);
209}
