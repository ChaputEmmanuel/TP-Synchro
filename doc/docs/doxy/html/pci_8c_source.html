75uint32_t PCILireLong(uint8_t bus, uint8_t slot, uint8_t fonction, uint8_t offset)
76{
77 uint32_t result;
78 uint32_t adresse = (((uint32_t)bus) << 16)
79 | (((uint32_t)slot) << 11)
80 | (((uint32_t)fonction) << 8)
81 | (((uint32_t)offset) & 0xFC)
82 | ((uint32_t)0x80000000);
83
84 // On envoie l'adresse qui nous intéresse
85 outl((uint16_t)PCI_CONFIG_ADDRESS, adresse);
86
87 // On y lit un long conformément à [1]
88 inl((uint16_t)PCI_CONFIG_DATA, result);
89
90 return result;
91}

93uint16_t PCILireMot(uint8_t bus, uint8_t slot, uint8_t fonction, uint8_t offset)
94{
95 uint16_t result;
96
97 // Si l'offset nous fait pointer sur le mot de poids fort, on décale
98 // à gauche d'un mot (2 * 8). Sinon non (0 * 8). Puis on masque car
99 // on ne veut qu'un mot
100 result = (uint16_t)(PCILireLong(bus, slot, fonction,
101 offset & ((uint16_t)~0x3)) >> ((offset & 2) * 8))& 0xFFFF;
102
103 return result;
104}

122void PCIEnumerationDesEquipementsDuBus(uint8_t bus)
123{
124 uint8_t numEquipement; // Pour boucler
125 uint16_t idVendeur;
126 uint16_t idEquipement;
127 uint32_t adresse; // Pour chercher les adresse d'E/S
128 uint8_t bar; // Parcourir les BAR
129
130 for (numEquipement = 0; numEquipement < PCI_NB_EQUIP_PAR_BUS; numEquipement++) {
131 //printk(PRINTK_CRITIQUE"id/nb %3d/%3d\r", numEquipement, PCI_NB_EQUIP_PAR_BUS);
132
133 // Lecture des numéros de vendeur et d'équipement
134 idVendeur = PCILireMot(bus, numEquipement, 0, PCI_ID_VENDEUR);
135
136 if (idVendeur != PCI_AUCUN_VENDEUR) {
137 if (PCINombreEquipements >= MANUX_NB_MAX_EQUIPEMENTS_PCI) {
138 printk(PRINTK_CRITIQUE"Trop d'equipements PCI pour moi (%d vs %d)!\n",
139 PCINombreEquipements, MANUX_NB_MAX_EQUIPEMENTS_PCI);
140 return;
141 }
142 idEquipement = PCILireMot(bus, numEquipement, 0, PCI_ID_EQUIPEMENT);
143
144 // Les identifiants
145 PCIEquipements[PCINombreEquipements].vendeur = idVendeur;
146 PCIEquipements[PCINombreEquipements].equipement = idEquipement;
147
148 PCIEquipements[PCINombreEquipements].configAddress.nul = 0 ;
149 PCIEquipements[PCINombreEquipements].configAddress.registreOffset = 0 ;
150 PCIEquipements[PCINombreEquipements].configAddress.numeroFonction = 0 ;
151 PCIEquipements[PCINombreEquipements].configAddress.numeroEquipement = numEquipement ;
152 PCIEquipements[PCINombreEquipements].configAddress.numeroBus = 0 ;
153
154
155 // L'interruption
156 PCIEquipements[PCINombreEquipements].interruption =
157 PCILireOctet(bus, numEquipement, 0, PCI_INTERRUPTION);
158
159 // On cherche l'adresse d'E/S
160 // On parcourt tous les registres mémoire
161 for (bar = PCI_BAR_0; bar <= PCI_BAR_5; bar += 4) {
162 adresse = PCILireLong(0, numEquipement, 0, bar);
163 // Est-ce une adresse d'E/S ?
164 if (adresse & PCI_BAR_IO) {
165 // On est aligné sur 32 bits, les 2 bits de poids faible sont donc
166 // non signifiants
167 PCIEquipements[PCINombreEquipements].adresseES = adresse & (~0x3);
168 } else {
169 printk_debug(DBG_KERNEL_A_FAIRE, "Memory BARs a prendre en compte !\n");
170 }
171 }
172
173 printk_debug(DBG_KERNEL_PCI, "[%d] Vd/Id 0x%x/0x%x\n",
174 PCINombreEquipements, idVendeur, idEquipement);
175 PCINombreEquipements++;
176 }
177 }
178}
