38int vsnprintf(char * str, const size_t l, char * format, va_list argList)
39{
40 size_t indice = 0;
41 // long long int n; // valeur associée à un %[l]d
42 long int n; // valeur associée à un %[l]d
43 char nombre[10]; // chaîne du nombre
44 char * s; // valeur associée à un %s
45 char c; // Affichage d'un caractère
46 int in; // indice pour les boucles internes
47 int nbChiffres; // pour les %[n]d
48 int base; // de l'affichage entier
49 int prefixe; // 0 int, 1 long int, 2 long long int
50
51 while ((*format) && (indice < l-1)) {
52 switch (*format) {
53 case '%' :
54 format++;
55 // Lecture de la taille
56 nbChiffres = 0;
57 while ((*format <= '9') && (*format >= '0')) {
58 nbChiffres = nbChiffres * 10 + *format - '0';
59 format++;
60 }
61
62 // Pas forcément bien là, mais j'ai pas mieux dans l'immédiat
63 prefixe = 0;
64 if (*format == 'l') {
65 format++;
66 if (*format == 'l') {
67 format++;
68 prefixe = 2;
69 } else {
70 prefixe = 1;
71 }
72 }
73 switch (*format) {
74 case 'o' :
75 base = 8;
76 goto affent;
77 case 'x' :
78 base = 16;
79 goto affent;
80 case 'd' :
81 base = 10;
82affent : switch (prefixe) {
83 case 2 :
84 n = 0; //;va_arg(argList, long long int); WARNING
85 break;
86 case 1 :
87 n = va_arg(argList, long int);
88 break;
89 default :
90 n = va_arg(argList, int);
91 break;
92 }
93 if (n < 0) {
94 n = -n;
95 str[indice++] = '-';
96 }
97 in = 0;
98 do {
99 nombre[in++] = chiffre[n%base];
100 n = n/base; // n est défini comme long long,
101 } while (n != 0);
102 while (nbChiffres > in) {
103 nbChiffres--;
104 str[indice++] = ' ';
105 }
106 do {
107 str[indice++] = nombre[--in];
108 } while (in);
109 break;
110 case 's' :
111 s = va_arg(argList, char *);
112 in = 0;
113 while (s[in]) {
114 str[indice++] = s[in++];
115 }
116 break;
117 case 'c' :
118 c = (char)va_arg(argList, int);
119 in = 0;
120 str[indice++] = c;
121 break;
122 default :
123 break;
124 }
125 break;
126 default :
127 str[indice++] = *format;
128 break;
129 }
130 format++;
131 }
132
133 str[(indice < l)?indice:(l-1)] = 0;
134
135 return indice;
136}
