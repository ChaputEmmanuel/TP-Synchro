93static inline uint16_t ordreDe(uint16_t taille)
94{
95 uint16_t result = MANUX_KMALLOC_ORDRE_MAX;
96 uint16_t tailleTotale = taille + sizeof(enteteBlocMemoire);
97
98 assert(tailleTotale <= (1<<MANUX_KMALLOC_ORDRE_MAX));
99
100 while (1<< (result - 1) >= tailleTotale) {
101 result --;
102 }
103
104 return result;
105}

139void * kmalloc(size_t n)
140{
141 enteteBlocMemoire * enteteBlocAlloue = NULL;
142 uint16_t ordre; //< Log2 de la taille totale
143 uint16_t o; //< indice de recherche de blocs
144
145 ordre = ordreDe(n);
146 if (ordre < MANUX_KMALLOC_ORDRE_MIN) {
147 ordre = MANUX_KMALLOC_ORDRE_MIN;
148 };
149
150 assert(ordre <= MANUX_KMALLOC_ORDRE_MAX);
151 assert(ordre >= MANUX_KMALLOC_ORDRE_MIN);
152
153 // On recherche un bloc assez gros pour la demande
154 o = ordre;
155 while (( o <= MANUX_KMALLOC_ORDRE_MAX) && (blocsLibres[o] == NULL)) {
156 o++;
157 }
158
159 assert(ordre <= MANUX_KMALLOC_ORDRE_MAX);
160 assert(ordre >= MANUX_KMALLOC_ORDRE_MIN);
161
162 // Si on a dépasser la taille max, c'est qu'on a rien trouvé. On
163 // alloue alors un bloc de taille maximale.
164 if (o > MANUX_KMALLOC_ORDRE_MAX) {
165 // WARNING traiter cas où MANUX_KMALLOC_ORDRE_MAX est plus gros qu'une page ...
166 blocsLibres[MANUX_KMALLOC_ORDRE_MAX] = allouerPage();
167 if (blocsLibres[MANUX_KMALLOC_ORDRE_MAX] == NULL) {
168 return NULL;
169 } else {
170#ifdef MANUX_KMALLOC_STAT
171 nbPagesAllouees++;
172#endif
173 o = MANUX_KMALLOC_ORDRE_MAX;
174 blocsLibres[o]->e.suivant = NULL;
175 }
176 }
177
178 // S'il est trop gros on le coupe en deux jusqu'à la bonne taille
179 while (o > ordre) {
180 decouperUnBloc(o);
181 o--;
182 }
183
184 // On y est, par construction il y a un bloc de la bonne taille
185 // on l'alloue
186 enteteBlocAlloue = blocsLibres[o];
187 blocsLibres[o] = enteteBlocAlloue->e.suivant;
188 enteteBlocAlloue->e.alloue.magique = MANUX_KMALLOC_MAGIQUE;
189 enteteBlocAlloue->e.alloue.ordre = ordre;
190
191 if (enteteBlocAlloue == NULL) {
192 return NULL;
193 } else {
194#ifdef MANUX_KMALLOC_STAT
195 nbAlloc[ordre]++;
196#endif
197
198 return enteteBlocAlloue + 1;
199 }
200}

211void reintegrerBloc(enteteBlocMemoire * bloc)
212{
213 uint16_t ordre = bloc->e.alloue.ordre; //< Ordre du bloc à intégrer
214 enteteBlocMemoire * siamois, *prec = NULL;
215
216 do {
217 assert(ordre <= MANUX_KMALLOC_ORDRE_MAX);
218 assert(ordre >= MANUX_KMALLOC_ORDRE_MIN);
219
220 // On cherche le siamois, dont l'adresse ne diffère que par le
221 // bit de l'ordre
222 prec = NULL;
223 siamois = blocsLibres[ordre];
224 while ((siamois != NULL) && (((uint32_t)siamois ^ (1<<ordre)) != (uint32_t)bloc)) {
225 prec = siamois;
226 siamois = siamois->e.suivant;
227 }
228
229 // Si le siamois était dans les libres, on l'enlève et on procède
230 // à la fusion
231 if (siamois != NULL) {
232 // On extrait le siamois
233 if (prec == NULL){ // Si c'était le premier
234 blocsLibres[ordre] = siamois->e.suivant;
235 } else { // Sinon
236 prec->e.suivant = siamois->e.suivant;
237 }
238
239 // On recrée le bloc constitué de la fusion. L'adresse est celle
240 // des deux avec le bit d'ordre ordre à zéro
241 bloc = (enteteBlocMemoire *)(((uint32_t)bloc)&(~(1 << ordre)));
242
243 // On remonte donc d'un ordre
244 ordre++;
245 }
246 } while ((siamois != NULL) // Si la fusion a pu être faite, on
247 &&(ordre < MANUX_KMALLOC_ORDRE_MAX)); // essaye de fusionner un ordre au dessus
248
249 // A ce stade, on a effectué toutes les fusions possibles
250 // concernant ce bloc. L'état final du bloc intègre ces
251 // éventuellles fusions et l'ordre a donc été incrémenté à chaque
252 // fusion. Il ne nous reste donc plus qu'à intègrer le bloc final
253 // dans les libres
254 bloc->e.suivant = blocsLibres[ordre];
255 blocsLibres[ordre] = bloc;
256
257}

265void kfree(void * p)
266{
267 enteteBlocMemoire * bloc = (enteteBlocMemoire *)p ;
268
269 bloc--;
270
271 if (bloc->e.alloue.magique != MANUX_KMALLOC_MAGIQUE) {
272 paniqueNoyau("Tentative de libérer un bloc 0x%x inconnu (m=0x%x, o=%d)\n",
273 p, bloc->e.alloue.magique, bloc->e.alloue.ordre);
274 }
275
276#ifdef MANUX_KMALLOC_STAT
277 nbFree[bloc->e.alloue.ordre]++;
278#endif

281 reintegrerBloc(bloc);
282 }
