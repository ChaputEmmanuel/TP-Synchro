117int virtioConsoleInitPeripherique(int PCINumeroPeripherique)
118{
119 PCIEquipement * pciEquip = PCIEquipementNumero(PCINumeroPeripherique);
120 void * pointeur; // Pour allouer de la mémoire en
121 // réception
122 VirtioFileVirtuelle * fr; // réception
123 // On va construire un tableau de buffers (et un de leurs
124 // longueurs) pour passer à virtioFournirBuffers
125 void * buffers[1];
126 int longueurs[1];
127
128 printk_debug(DBG_KERNEL_VIRTIO, "IN\n");
129
130 // On renseigne la structure
131 virtioInitPeripheriquePCI(&(virtioConsole.virtioPeripherique),
132 PCINumeroPeripherique,
133 VIRTIO_CONSOLE_F_SIZE
134 | VIRTIO_CONSOLE_F_MULTIPORT);
135
136 // La suite est à mettre dans virtio (ou à virer ?)
137 if (i8259aAjouterHandler(pciEquip->interruption,
138 virtioConsoleGestionInt,
139 &virtioConsole)){
140 printk(PRINTK_CRITIQUE "Trop de handler sur l'interruption %d\n", pciEquip->interruption);
141 paniqueNoyau("Pas possible");
142 return ENOENT;
143 }
144 i8259aAutoriserIRQ(pciEquip->interruption);
145 virtioConsole.nbItRecues = 0;
146
147 // On ajoute des buffers sur la file d'entrée
148 fr = &(virtioConsole.virtioPeripherique.filesVirtuelles[VIRTIO_CONSOLE_PORT0_IN]);
149
150 // Pas de malloc, ... WARNING : faire version avec malloc
151 pointeur = allouerPages(NB_PAGES(fr->taille*(1024)));
152
153 for (int i = 0 ; i < fr->taille ; i++) {
154 buffers[0] = pointeur;
155 longueurs[0] = 1024;
156 pointeur += longueurs[0];
157
158 virtioFournirBuffers(&(virtioConsole.virtioPeripherique),
159 VIRTIO_CONSOLE_PORT0_IN,
160 buffers, longueurs,
161 1,
162 VRING_DESC_F_WRITE);
163 }
164
165 printk_debug(DBG_KERNEL_VIRTIO, "OUT\n");
166
167 return ESUCCES;
168}
